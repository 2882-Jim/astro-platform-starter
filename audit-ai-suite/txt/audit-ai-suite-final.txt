// ---------- Saving (only in this browser, under names starting "aiSuite.") ----------
const STORE = "aiSuite.";
const memory = {};          // a copy in memory, so the page still works if the browser blocks saving
let storageWorks = true;

try { localStorage.setItem(STORE + "test", "1"); localStorage.removeItem(STORE + "test"); }
catch (e) { storageWorks = false; }

function load(key, fallback) {      // read a saved value; never breaks the page
  if (key in memory) return memory[key];
  try {
    const text = localStorage.getItem(STORE + key);
    const value = text === null ? fallback : JSON.parse(text);
    memory[key] = value;
    return value;
  } catch (e) { storageWorks = false; return fallback; }
}

function save(key, value) {         // save a value; returns false if the browser wouldn't save it
  memory[key] = value;
  try { localStorage.setItem(STORE + key, JSON.stringify(value)); return true; }
  catch (e) { storageWorks = false; return false; }
}

function clearSaved() {             // delete everything this app saved in this browser
  Object.keys(memory).forEach(k => delete memory[k]);
  try {
    Object.keys(localStorage).filter(k => k.startsWith(STORE)).forEach(k => localStorage.removeItem(k));
    return true;
  } catch (e) { storageWorks = false; return false; }
}

// Another tab of this app saved or cleared something: forget the copy in memory, so the next read and save use
// what's saved now (no redraw, so nothing half-typed here is lost)
window.addEventListener("storage", e => {
  if (!e.key || (e.key.startsWith(STORE) && e.key !== STORE + "test")) Object.keys(memory).forEach(k => delete memory[k]);
});

function loadList(key) {            // a saved list, or an empty one if it's missing or damaged
  const list = load(key, []);
  return Array.isArray(list) ? list : [];
}

// ---------- Text and labels ----------
function safe(text) {               // stops typed or loaded text being treated as page code
  return String(text == null ? "" : text).replace(/&/g, "&amp;").replace(/</g, "&lt;")
    .replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}

function label(rating) {            // a coloured rating label with the word on it
  return RATINGS[rating] ? `<span class="label ${RATINGS[rating].css}">${rating}</span>` : safe(rating);
}

function options(list, selected) {  // turns a list into dropdown options (value = position in the list)
  return list.map((item, i) => `<option value="${i}"${i === selected ? " selected" : ""}>${safe(item)}</option>`).join("");
}

function cut(text, max) {           // trims typed text to a sensible length
  return typeof text === "string" ? text.trim().slice(0, max) : "";
}

function matches(text, typed) {     // search: ignores capitals; every word typed must appear
  const words = String(typed).toLowerCase().split(/\s+/).filter(w => w);
  return words.length > 0 && words.every(w => text.toLowerCase().includes(w));
}

function privacyHint(text) {        // a gentle warning if the text might name someone
  const t = String(text || "");
  const word = new RegExp("\\b(" + PRIVACY_WORDS.join("|") + ")\\b", "i");
  return (/\S+@\S+/.test(t) || word.test(t) || /\d{8,}/.test(t))
    ? "This might include a name, email address or account number. Remove anything that identifies a client or a person." : "";
}

// ---------- Dates (saved as YYYY-MM-DD, shown as 6 Nov 2026) ----------
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
function pad(n) { return String(n).padStart(2, "0"); }
function todayISO() { const d = new Date(); return d.getFullYear() + "-" + pad(d.getMonth() + 1) + "-" + pad(d.getDate()); }
function dayNo(iso) { const p = iso.split("-").map(Number); return Date.UTC(p[0], p[1] - 1, p[2]) / 864e5; }  // whole days since 1970
function addDays(iso, n) { const d = new Date((dayNo(iso) + n) * 864e5); return d.getUTCFullYear() + "-" + pad(d.getUTCMonth() + 1) + "-" + pad(d.getUTCDate()); }
function isISO(s) { return typeof s === "string" && /^\d{4}-\d{2}-\d{2}$/.test(s) && addDays(s, 0) === s; }
function daysBetween(a, b) { return dayNo(b) - dayNo(a); }
function weekNo(iso) { return Math.floor((dayNo(iso) - dayNo("2026-01-05")) / 7); }   // weeks since Monday 5 Jan 2026
function formatDate(iso) { if (!isISO(iso)) return ""; const p = iso.split("-").map(Number); return p[2] + " " + MONTHS[p[1] - 1] + " " + p[0]; }
function newId(prefix) { return prefix + "-" + Math.random().toString(36).slice(2, 10) + Math.random().toString(36).slice(2, 6); }   // random, so it holds no time

function verdictFor(task, lvl, where) {   // what someone at this level should do with this task
  const rating = task.ratings[where], need = LEVELS.indexOf(task.minLevel);
  if (rating === "Not permitted") return { key: "notpermitted", say: "Not permitted",
    why: task.ratings.cloud === "Not permitted" ? "Don't use AI for this." : "Not with a personal account: use a firm-approved tool." };
  if (lvl < need) return { key: "notyet", say: need === 3 ? "Not yet, ask your manager" : "Not yet, ask your senior", why: "Needs " + task.minLevel + " or above." };
  if (lvl <= HAND_FIRST_UP_TO[rating]) return { key: "byhand", say: "Do it by hand first",
    why: rating === "Significant" ? "High risk: do it yourself first, and ask your manager before using AI."
       : rating === "Low" ? "Try it yourself first, then compare with AI's answer." : "So you can tell when AI gets it wrong." };
  return { key: "goahead", say: "Go ahead", why: "Follow the rating's 'what to do'." };
}

function checkerRating(answers, where) {  // the checker's three yes/no answers (plus the switch) give a rating
  const clientData = answers[0], onFile = answers[1], couldMatter = answers[2];
  if (where === "personal" && clientData) return { rating: "Not permitted", reason: "Client or personal data must never go into a personal AI account." };
  if (where === "personal" && couldMatter) return { rating: "Not permitted", reason: "Risky work needs a firm-approved, tested tool, and a personal account isn't one." };
  if (couldMatter) return { rating: "Significant", reason: "A mistake could matter and might be missed, so the tool must be approved and tested for this." };
  if (onFile) return { rating: "Medium", reason: "It goes on file or is relied on, so every item must be checked to source." };
  if (!clientData) return { rating: "Low", reason: "It only helps you think or draft, and nothing relies on it." };
  return { rating: "Low", reason: where === "local" ? "Nothing relies on it. Client data is only fine here if the firm has approved this local tool."
                                                  : "Nothing relies on it. Client data is only fine here because the tool is firm-approved." };
}

function cleanReport(r) {
  if (!r || typeof r !== "object") return null;
  const type = OLD_MISTAKE_TYPES[r.type] || r.type;   // older files used other names for the types
  if (!SECTIONS.includes(r.section) || !MISTAKE_TYPES.includes(type)) return null;
  const wrong = cut(r.wrong, 300);
  if (!wrong) return null;
  return { id: typeof r.id === "string" && r.id.length <= 40 ? r.id : (r.date + r.section + type + wrong).slice(0, 40),
           date: isISO(r.date) ? r.date : todayISO(), section: r.section, type, wrong,
           caught: cut(r.caught, 200), impact: cut(r.impact, 200),
           context: r.context === "practice" || r.context === "real" ? r.context : "",
           task: TASKS.some(t => t.task === r.task) ? r.task : "", example: r.example === true };
}
function loadReports() { return loadList("reports").map(cleanReport).filter(Boolean); }

function loadSteps() { const s = load("steps", null); return Array.isArray(s) && s.length === START_STEPS.length ? s.map(v => v === true) : START_STEPS.map(() => false); }

function validPicks(p) {

    p.every((v, i) => Number.isInteger(v) && v >= -1 && v < PITFALLS[i].sentences.length) ? p : null;
}

function makeExport(madeWith) {    // madeWith = "basic" or "final"
  const out = { app: EXPORT_APP, version: EXPORT_VERSION, madeWith, exportedOn: todayISO(), exportedAt: new Date().toISOString(), anonId: getAnonId(),
    level: load("levelSet", false) ? (LEVELS[load("level", 0)] || "") : "",
    reports: loadReports().filter(r => !r.example).map(r => ({ id: r.id, date: r.date, section: r.section, type: r.type,
      wrong: r.wrong, caught: r.caught, impact: r.impact, context: r.context, task: r.task })) };
  if (madeWith === "final") {
    out.steps = loadSteps();
    out.game = { picks: loadPicks() };
    out.settingsCopiedOn = load("settingsCopiedOn", "");
    out.notes = loadList("notes").map(cleanNote).filter(Boolean);
    out.pilots = loadList("pilots").map(cleanPilot).filter(Boolean);
  }
  const team = cleanTeam(load("team", ""));
  if (team) out.team = team;       // optional: a team or office, never a name
  return out;
}

function cleanNote(n) {            // an AI use note: reviewer role only, never a name

function cleanPilot(p) {           // a pilot: whole numbers in sensible ranges, or it's skipped
  const whole = (v, lo, hi) => Number.isInteger(v) && v >= lo && v <= hi;
  if (!p || typeof p !== "object" || !whole(p.baselineMinutes, 0, 600) || !whole(p.aiMinutes, 0, 600) ||
      !whole(p.errorsCaught, 0, 1000) || !whole(p.startedWith, 1, 500) || !whole(p.stillUsing, 0, p.startedWith)) return null;
  return { id: typeof p.id === "string" ? p.id.slice(0, 40) : "", goal: cut(p.goal, 150),
           tool: APPROVED_TOOLS.some(t => t.name === p.tool) ? p.tool : "Other",
           baselineMinutes: p.baselineMinutes, aiMinutes: p.aiMinutes, errorsCaught: p.errorsCaught,
           startedWith: p.startedWith, stillUsing: p.stillUsing };
}

function normaliseExport(obj) {    // check a loaded file; bad parts are skipped, never trusted
  if (!obj || typeof obj !== "object") return { ok: false, problem: "This isn't a file from this app." };
  if (obj.app !== EXPORT_APP) return { ok: false, problem: "This file wasn't made by this app." };
  if (!Number.isInteger(obj.version) || obj.version < 1) return { ok: false, problem: "This file's version isn't recognised." };
  if (typeof obj.anonId !== "string" || !/^[a-z0-9-]{4,40}$/.test(obj.anonId)) return { ok: false, problem: "This file is missing its reference code." };
  let skipped = 0;
  const clean = (list, max, fn) => {
    if (!Array.isArray(list)) return null;
    const out = list.slice(0, max).map(fn).filter(Boolean);
    skipped += list.length - out.length;   // bad items and any past the limit
    return out;
  };
  const reports = clean(obj.reports, 1000, r => { const c = cleanReport(r); return c && !c.example ? c : null; }) || [];
  const notes = clean(obj.notes, 500, cleanNote), pilots = clean(obj.pilots, 50, cleanPilot);   // before 'skipped' is used below
  const steps = Array.isArray(obj.steps) && obj.steps.length === START_STEPS.length && obj.steps.every(v => typeof v === "boolean") ? obj.steps : null;
  return { ok: true, skipped, newer: obj.version > EXPORT_VERSION, person: {
    anonId: obj.anonId, madeWith: obj.madeWith === "final" ? "final" : "basic",
    exportedOn: isISO(obj.exportedOn) ? obj.exportedOn : "",
    exportedAt: typeof obj.exportedAt === "string" && /^\d{4}-\d{2}-\d{2}T[\d:.]{5,12}Z$/.test(obj.exportedAt) ? obj.exportedAt : "", level: LEVELS.includes(obj.level) ? obj.level : "",
    reports, steps, picks: validPicks(obj.game && obj.game.picks),
    settingsCopiedOn: isISO(obj.settingsCopiedOn) ? obj.settingsCopiedOn : ("settingsCopiedOn" in obj ? "" : null),
    notes, pilots, team: cleanTeam(obj.team), group: "" } };
}

function readExport(text) {        // text from a file -> checked people (or a plain-English problem)
  if (typeof text !== "string" || text.length > 1000000) return { ok: false, problem: "This file is too big to be from this app." };
  let obj;
  try { obj = JSON.parse(text); }
  catch (e) { return { ok: false, problem: "This file couldn't be read. Is it the file from 'Send to my manager'?" }; }
  if (!obj || typeof obj !== "object" || obj.kind !== "group") {   // one person's file
    const r = normaliseExport(obj);
    if (r.ok) r.people = [r.person];
    return r;
  }
  // A group file: { app, version, kind: "group", name, people: [one person's file, ...] }. Each person is checked on their own.
  if (obj.app !== EXPORT_APP) return { ok: false, problem: "This file wasn't made by this app." };
  const name = cleanTeam(obj.name) || "Unnamed group";
  if (!Array.isArray(obj.people)) return { ok: false, problem: "This group file has no people in it." };
  const checked = obj.people.slice(0, 500).map(normaliseExport), list = checked.filter(r => r.ok).map(r => r.person);
  if (!list.length) return { ok: false, problem: "None of the people in this group file could be read." };
  list.forEach(p => { p.group = name; });
  return { ok: true, group: name, people: list, newer: obj.version > EXPORT_VERSION || checked.some(r => r.newer),
           skipped: checked.reduce((n, r) => n + (r.ok ? r.skipped : 1), 0) + Math.max(0, obj.people.length - 500) };
}

function mergePeople(people) {     // one entry per person: each part comes from their latest file that has it
  const byId = Object.create(null);   // a plain lookup, so codes such as 'constructor' are safe
  people.forEach(p => (byId[p.anonId] = byId[p.anonId] || []).push(p));
  return Object.values(byId).map(files => {
    files.sort((a, b) => (a.exportedAt || a.exportedOn || "").localeCompare(b.exportedAt || b.exportedOn || ""));   // oldest file first
    const latest = key => { for (let i = files.length - 1; i >= 0; i--) if (files[i][key] !== null && files[i][key] !== undefined) return files[i][key]; return null; };
    const seen = Object.create(null), reports = [];
    files.forEach(f => f.reports.forEach(r => { if (!seen[r.id]) { seen[r.id] = true; reports.push(r); } }));
    return { anonId: files[0].anonId, madeWith: files.some(f => f.madeWith === "final") ? "final" : "basic",
             exportedOn: files[files.length - 1].exportedOn, level: latest("level") || "", reports,
             steps: latest("steps"), picks: latest("picks"), settingsCopiedOn: latest("settingsCopiedOn"),
             notes: latest("notes"), pilots: latest("pilots"), team: latest("team") || "", group: latest("group") || "",
             source: latest("source") || "", files: files.length };
  });
}

function countReports(reports) {    // how many reports in each section x mistake type square
  const counts = SECTIONS.map(() => MISTAKE_TYPES.map(() => 0));
  reports.forEach(r => counts[SECTIONS.indexOf(r.section)][MISTAKE_TYPES.indexOf(r.type)]++);
  return counts;
}
function topSquares(counts) {       // the busiest square(s): { max, names }
  let max = 0, names = [];
  counts.forEach((row, s) => row.forEach((n, t) => {
    const name = SECTIONS[s] + " / " + MISTAKE_TYPES[t];
    if (n > max) { max = n; names = [name]; } else if (n === max && n > 0) names.push(name);
  }));
  return { max, names };
}
function heatTable(counts, pickFn, prefix) { // the grid: each square is a button; darker means more reports
  prefix = prefix || "cell";        // the start of each square's id, so two maps on one page don't clash
  const max = topSquares(counts).max;
  let html = `<table class="grid heat"><caption>AI mistakes by audit section and type</caption>
    <tr><th scope="col">Section</th>${MISTAKE_TYPES.map(t => `<th scope="col">${t}</th>`).join("")}</tr>`;
  counts.forEach((row, s) => {
    html += `<tr><th scope="row">${SECTIONS[s]}</th>` + row.map((n, t) =>
      `<td class="cellwrap h${n ? Math.ceil(4 * n / max) : 0}"><button class="cell" id="${prefix}-${s}-${t}" data-s="${s}" data-t="${t}"
        aria-label="${SECTIONS[s]}, ${MISTAKE_TYPES[t]}: ${n} report${n === 1 ? "" : "s"}" onclick="${pickFn}(${s}, ${t})">${n}</button></td>`).join("") + `</tr>`;
  });
  return `<div class="scroll">${html}</table></div>`;
}

async function copyText(boxId, msgId, onCopied) {   // copy a box's text; if the browser blocks it, select the text instead
  const box = document.getElementById(boxId), msg = document.getElementById(msgId);
  try { await navigator.clipboard.writeText(box.value); msg.textContent = "Copied."; }
  catch (e) { box.focus(); box.select(); box.setSelectionRange(0, box.value.length); msg.textContent = "Press Ctrl+C (or Cmd+C on a Mac) to copy."; }

}

function downloadFile(name, text) { // save a file to the computer; returns false if the browser blocks it
  try {
    const url = URL.createObjectURL(new Blob([text], { type: "application/json" }));
    const a = document.createElement("a");
    a.href = url; a.download = name;
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 10000);
    return true;
  } catch (e) { return false; }
}

// ---------- Data check: lists any typos in the data above, in one box on the page ----------
function checkData() {
  const problems = [];
  TASKS.forEach(t => {
    const r = t.ratings || {};
    if (!SECTIONS.includes(t.section) || !RATINGS[r.personal] || !RATINGS[r.cloud] || !RATINGS[r.local] || !LEVELS.includes(t.minLevel)
        || (t.recipe !== "" && !PROMPT_RECIPES.some(p => p.id === t.recipe)) || !PITFALLS.some(p => p.id === t.spot)) problems.push("Task: " + t.task);
  });
  PITFALLS.forEach(p => {
    if (!MISTAKE_TYPES.includes(p.type) || !(p.wrong >= 0 && p.wrong < p.sentences.length) || !p.terms.every(term => GLOSSARY.some(g => g.term === term))) problems.push("Spot the mistake: " + p.name);
  });
  APPROVED_TOOLS.forEach(t => { if (!SWITCH_OPTIONS[t.where] || !TOOL_STATUSES[t.status]) problems.push("Tool: " + t.name); });
  if (PROMPT_PRESETS.length !== LEVELS.length) problems.push("PROMPT_PRESETS needs one line per level");
  PROMPT_RECIPES.forEach(p => { if (!p.id || !p.name || !Array.isArray(p.inputs) || !p.checks.every(c => PROMPT_CHECKS[c])) problems.push("Prompt recipe: " + p.name); });
  GLOSSARY.forEach(g => { if (!g.term || !g.means || !g.why) problems.push("Glossary: " + g.term); });
  QUIZ_BANK.forEach((q, i) => {
    const card = q.pitfall ? PITFALLS.find(p => p.id === q.pitfall) : q, task = TASKS.find(t => t.task === q.task);
    const good = LEVELS.includes(q.minLevel) && typeof q.explain === "string" && q.explain && (
      q.type === "spot" ? !!card && Array.isArray(card.sentences) && card.wrong >= 0 && card.wrong < card.sentences.length && (q.pitfall || (q.check && q.name && q.task)) && card.terms.every(term => GLOSSARY.some(g => g.term === term))
      : q.type === "rate" ? !!SWITCH_OPTIONS[q.where] && !!q.q && (task ? !!task.ratings[q.where] : Array.isArray(q.answers) && q.answers.length === 3)
      : q.type === "rightwrong" ? typeof q.good === "boolean" && !!q.q : false);
    if (!good) problems.push("Quiz question " + (i + 1));
  });
  EXAMPLE_REPORTS.forEach((e, i) => { if (!SECTIONS[e[0]] || !MISTAKE_TYPES[e[1]]) problems.push("Example report " + (i + 1)); });
  return problems.length ? `<div class="error" role="alert"><b>Check the spelling in the data:</b><ul>${problems.map(p => `<li>${safe(p)}</li>`).join("")}</ul></div>` : "";
}

  let html = "";

function recipeBlocked(r) { const t = TASKS[pb.task]; return !!t && t.recipe === r.id && t.ratings[where] === "Not permitted"; }   // the banner's task is Not permitted here

function recipeHtml(r, k) { // one recipe card: a fold-out with 1 or 2 boxes, the prompt, and Copy

function recipePrompt(r) {  // the task (with what was typed), the built-in checks, then the pasted text
  const fill = (text) => text.replace(/\{(\w+)\}/g, (m, id) => val(r, id).trim() || "[" + r.inputs.find(x => x[0] === id)[1].toLowerCase() + "]");
  const rules = r.checks.map(c => PROMPT_CHECKS[c]).concat([SETTINGS_RULES]);
  let text = "Task: " + fill(r.task) + "\n\nRules:\n" + rules.map(x => "- " + x).join("\n");
  if (r.inputs.some(x => x[0] === "paste")) text += "\n\n--- TEXT START ---\n" + (val(r, "paste").trim() || "[paste here]") + "\n--- TEXT END ---";
  return text;
}

// =====================================================================
function quizCard(q) { return q.pitfall ? PITFALLS.find(p => p.id === q.pitfall) : q; }   // a "spot" question's card
function quizOptions(q) {   // what you can pick: [value, words]
  if (q.type === "spot") return quizCard(q).sentences.map((s, i) => [i, s]);
  if (q.type === "rate") return Object.keys(RATINGS).map(r => [r, r]);
  return [[true, "Yes, good use"], [false, "No, not good use"]];
}
function quizRight(q) {     // the place of the right answer in quizOptions (worked out, so it always agrees with the task list)
  const t = TASKS.find(x => x.task === q.task);
  const right = q.type === "spot" ? quizCard(q).wrong : q.type === "rate" ? (t ? t.ratings[q.where] : checkerRating(q.answers, q.where).rating) : q.good;
  return quizOptions(q).findIndex(o => o[0] === right);
}
function quizTitle(q) { return q.type === "spot" ? quizCard(q).task : q.q; }
function quizScore() { return quiz.picks.filter((k, n) => k === quizRight(QUIZ_BANK[quiz.items[n]])).length; }

function makeQuiz(first) {  // 5 different questions at or below your level, in a random order (first = one to start with, or -1)
  const pool = QUIZ_BANK.map((q, i) => i).filter(i => LEVELS.indexOf(QUIZ_BANK[i].minLevel) <= level && i !== first);
  for (let i = pool.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [pool[i], pool[j]] = [pool[j], pool[i]]; }
  return { items: (first >= 0 ? [first] : []).concat(pool).slice(0, 5), picks: [], at: 0, done: false };
}
function startQuiz(first) { quiz = makeQuiz(first); go("quiz"); }

function quizScreen() {
  if (!quiz) quiz = makeQuiz(-1);
  if (quiz.done) return quizResult();
  const n = quiz.items.length, q = QUIZ_BANK[quiz.items[quiz.at]], pick = quiz.picks[quiz.at], done = pick !== undefined, right = quizRight(q);
  const dots = quiz.items.map((x, k) => k < quiz.picks.length ? (quiz.picks[k] === quizRight(QUIZ_BANK[x]) ? "✓" : "✗") : k === quiz.at ? "●" : "○");
  const card = q.type === "spot" ? quizCard(q) : null;
  const opts = quizOptions(q).map((o, k) => `<button class="sentence${done && k === right ? " answer" : ""}${k === pick ? " picked" : ""}" id="opt-${k}"
    onclick="pickQuiz(${k})"${done ? " disabled" : ""}>${q.type === "rate" ? label(o[1]) : o[1]}${done && k === right ? `<b class="mark">✓ ${card ? "This is the wrong sentence" : "Right answer"}</b>` : ""}${k === pick ? `<span class="mark-pick"> (your answer)</span>` : ""}</button>`).join("");
  const ask = { spot: "Spot the mistake: one sentence is wrong. Which one?", rate: "Pick the rating", rightwrong: "Right or wrong way: is this good AI use?" }[q.type];
  const links = card ? card.terms.map(term => GLOSSARY.findIndex(g => g.term === term)).filter(g => g >= 0)
    .map(g => `<button class="link glossary-link" id="gl-${g}" onclick="openTerm(${g})">See in glossary: ${GLOSSARY[g].term}</button>`).join(" ") : "";
  return `<h1>Quiz</h1>
    <p class="dots" aria-hidden="true">${dots.map(d => `<span>${d}</span>`).join("")}</p>
    <p class="small" id="quizCount">Question ${quiz.at + 1} of ${n} · ${quizScore()} right so far</p>
    <h2 id="qHead" tabindex="-1">${ask}</h2>
    ${card ? `<p><b>${card.task}</b></p>${card.source ? `<div class="source"><p class="small">Source (made-up):</p>${card.source}</div>` : ""}`
      : `<p class="scenario">${q.q}</p>${q.type === "rate" ? `<p class="small">Where the AI runs: <b>${SWITCH_OPTIONS[q.where].name}</b></p>` : ""}`}
    ${opts}
    ${done ? `<div class="box" id="feedback"><h3 id="verdict" tabindex="-1">${pick === right ? "✓ Right!" : "✗ Not quite."}</h3>
      ${pick === right ? "" : `<p><b>The answer:</b> ${quizOptions(q)[right][1]}</p>`}<p>${q.explain}</p>
      ${card ? `<p><b>Check that catches it:</b> ${card.check}</p>${links}` : ""}</div>
      <button class="btn" id="quizNext" onclick="nextQuestion()">${quiz.at === n - 1 ? "See my mark" : "Next question"}</button>` : ""}`;
}
function pickQuiz(k) {      // one go per question; the result shows straight away
  if (quiz.picks[quiz.at] !== undefined) return;
  quiz.picks[quiz.at] = k;
  const q = QUIZ_BANK[quiz.items[quiz.at]];
  if (q.pitfall) {          // the manager's Spot the mistake figures use these answers
    const picks = loadPicks(); picks[PITFALLS.findIndex(p => p.id === q.pitfall)] = k; save("picks", picks);
  }
  draw();
  focusId("verdict");       // the picked button is now disabled, so read out the result instead
}
function nextQuestion() {
  if (quiz.at < quiz.items.length - 1) { quiz.at++; draw(); focusId("qHead"); return; }
  quiz.done = true;
  const best = loadBest(), score = quizScore();
  if (best === null || score > best) save("quizBest", score);
  tick(3);                  // finishing a quiz ticks "Do the quiz"
  draw();
  focusId("mark");
}
function quizResult() {
  const score = quizScore(), n = quiz.items.length, best = loadBest();
  const recap = quiz.items.map((x, k) => { const q = QUIZ_BANK[x], right = quizRight(q), opts = quizOptions(q), card = q.type === "spot" ? quizCard(q) : null;
    return `<li><b>${quizTitle(q)}</b><br>${quiz.picks[k] === right ? "✓ Right" : "✗ Not quite"}: you said "${opts[quiz.picks[k]][1]}".
      ${quiz.picks[k] === right ? "" : `<br>Right answer: "${opts[right][1]}".`}<br><span class="small">${q.explain}${card ? " Check: " + card.check : ""}</span></li>`; }).join("");
  return `<h1>Quiz</h1>
    <h2 id="mark" tabindex="-1">You got ${score} of ${n}</h2>
    <p>${score === n ? "Excellent: full marks!" : score >= 3 ? "Good work." : "Worth another go: the answers below will help."}</p>
    <p class="small" id="bestLine">Your best: ${best} of 5</p>
    <ol class="recap">${recap}</ol>
    <button class="btn" id="quizAgain" onclick="startQuiz(-1)">Try 5 more</button>
    <button class="plain" id="backLearn" onclick="go('learn')">Back to Learn</button>
    <p><button class="link" id="printCheat" onclick="printCheat()">Print the AI pitfalls cheat sheet</button></p>`;
}
function printCheat() {    // every "spot the mistake" check on one page
  const best = loadBest(), cards = QUIZ_BANK.filter(q => q.type === "spot").map(quizCard);
  printBlock(`<h1>AI pitfalls cheat sheet</h1>${best === null ? "" : `<p>Your best quiz mark: ${best} of 5.</p>`}

function noteRating(taskName, toolName) {   // the rating for a task with a tool ("" if it can't be rated)
  const t = TASKS.find(x => x.task === taskName), tool = APPROVED_TOOLS.find(x => x.name === toolName);
  return t && tool ? t.ratings[tool.where] : "";
}
function noteNeed(rating, minutes) {   // the requirement rule: "required", "recommended" or "notpermitted"
  if (rating === "Not permitted") return "notpermitted";
  return rating === "Significant" || (rating === "Medium" && minutes !== null && minutes >= NOTE_REQUIRED_MINUTES) ? "required" : "recommended";
}
const NEED_WORDS = { required: "Note required", recommended: "Note recommended", notpermitted: "Not permitted: don't use AI for this with this tool" };
function wholeOrNull(v) { const n = Number(v); return v !== "" && Number.isInteger(n) && n >= 0 ? n : null; }

  const rating = t.ratings[tool.where], v = verdictFor(t, level, tool.where), need = noteNeed(rating, wholeOrNull(note.minutes));
  return `<p class="need need-${need}" id="noteNeed">${label(rating)} <b>${NEED_WORDS[need]}</b></p>

// The pilot tracker (one small test of a tool), inside "Running a small pilot?"
function liveLine() {       // the difference per task, while you type
  const b = wholeOrNull(pilot.baselineMinutes), a = wholeOrNull(pilot.aiMinutes);
  if (b === null || a === null) return "Fill in both times to see the difference.";
  return a < b ? "Minutes saved per task: " + (b - a) : a > b ? "AI was " + plural(a - b, "minute") + " slower per task" : "No difference per task.";
}
function pilotResult(p) {   // e.g. "Saved 10 minutes per task (25% faster)"
  const diff = p.baselineMinutes - p.aiMinutes, pct = p.baselineMinutes ? Math.round(100 * Math.abs(diff) / p.baselineMinutes) : null;
  if (diff > 0) return "Saved " + plural(diff, "minute") + " per task (" + pct + "% faster)";
  if (diff < 0) return "AI was " + plural(-diff, "minute") + " slower per task" + (pct === null ? "" : " (" + pct + "% slower)");
  return "No difference in time per task";
}

function savePilot() {      // checked with the same rules a manager's dashboard uses (cleanPilot)
  const num = v => v === "" ? NaN : Number(v);
  const raw = { id: newId("p"), goal: cut(pilot.goal, 150), tool: APPROVED_TOOLS[pilot.tool].name, baselineMinutes: num(pilot.baselineMinutes),
    aiMinutes: num(pilot.aiMinutes), errorsCaught: num(pilot.errorsCaught), startedWith: num(pilot.startedWith), stillUsing: num(pilot.stillUsing) };
  const ok = (v, lo, hi) => Number.isInteger(v) && v >= lo && v <= hi, clean = cleanPilot(raw);
  if (!raw.goal) problems.push("Say what you're testing (one line is fine).");
  if (!ok(raw.baselineMinutes, 0, 600)) problems.push("Minutes per task without AI: a whole number from 0 to 600.");
  if (!ok(raw.aiMinutes, 0, 600)) problems.push("Minutes per task with AI: a whole number from 0 to 600.");
  if (!ok(raw.errorsCaught, 0, 1000)) problems.push("AI mistakes caught: a whole number from 0 to 1,000.");
  if (!ok(raw.startedWith, 1, 500)) problems.push("People who started: a whole number from 1 to 500.");
  if (!ok(raw.stillUsing, 0, 500)) problems.push("People still using it: a whole number from 0 to 500.");
  else if (ok(raw.startedWith, 1, 500) && raw.stillUsing > raw.startedWith) problems.push("People still using it can't be more than the people who started.");
  if (!clean && !problems.length) problems.push("Please check the numbers.");
  if (problems.length) { problemsAt = "pilot"; draw(); return; }
  const list = loadPilots();
  list.unshift(clean);
  tell("pilot", save("pilots", list) ? "Pilot saved." : "Saved for now, but this browser isn't saving, so it will be lost when you close the page.");
  pilot = { goal: "", tool: pilot.tool, baselineMinutes: "", aiMinutes: "", errorsCaught: "", startedWith: "", stillUsing: "" };
  draw();
}

// Locked team files: AES-GCM, with a key made from the team code (PBKDF2-SHA256, random salt and IV)
// =====================================================================
const KDF_ROUNDS = 250000;
const CODE_LETTERS = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";   // no 0/O or 1/I, so codes are easy to read out
function canEncrypt() { return !!(window.crypto && window.crypto.subtle && window.crypto.getRandomValues); }
function cleanCode(c) { return String(c || "").toUpperCase().replace(/[^A-Z0-9]/g, ""); }   // spaces, dashes and capitals don't matter
function toB64(bytes) { let s = ""; for (let i = 0; i < bytes.length; i += 0x8000) s += String.fromCharCode.apply(null, bytes.subarray(i, i + 0x8000)); return btoa(s); }
function fromB64(text) { const s = atob(text), b = new Uint8Array(s.length); for (let i = 0; i < s.length; i++) b[i] = s.charCodeAt(i); return b; }
async function codeKey(code, salt, rounds) {
  const base = await crypto.subtle.importKey("raw", new TextEncoder().encode(cleanCode(code)), "PBKDF2", false, ["deriveKey"]);
  return crypto.subtle.deriveKey({ name: "PBKDF2", hash: "SHA-256", salt, iterations: rounds }, base, { name: "AES-GCM", length: 256 }, false, ["encrypt", "decrypt"]);
}
async function lockText(text, code) {   // -> the small wrapper that is saved as the file
  const salt = crypto.getRandomValues(new Uint8Array(16)), iv = crypto.getRandomValues(new Uint8Array(12));
  const data = await crypto.subtle.encrypt({ name: "AES-GCM", iv }, await codeKey(code, salt, KDF_ROUNDS), new TextEncoder().encode(text));
  return { app: EXPORT_APP, kind: "team-file", version: 1, cipher: "AES-GCM", kdf: "PBKDF2-SHA256", rounds: KDF_ROUNDS,
           salt: toB64(salt), iv: toB64(iv), data: toB64(new Uint8Array(data)) };
}
async function openTeamFile(text, code) {   // -> { ok, person } or { ok: false, problem } in plain English
  if (text.length > 5000000) return { ok: false, problem: "This file is too big to be a team file." };
  let w;
  try { w = JSON.parse(text); } catch (e) { return { ok: false, problem: "This isn't a team file (it couldn't be read)." }; }
  if (!w || typeof w !== "object" || w.app !== EXPORT_APP) return { ok: false, problem: "This file wasn't made by this app." };
  if (w.kind !== "team-file") return { ok: false, problem: "This is the no-names file for the management dashboard, not a team file." };
  let salt, iv, data;
  try { salt = fromB64(w.salt); iv = fromB64(w.iv); data = fromB64(w.data); } catch (e) { salt = null; }
  if (!salt || salt.length !== 16 || iv.length !== 12 || !Number.isInteger(w.rounds) || w.rounds < 100000 || w.rounds > 2000000)
    return { ok: false, problem: "This team file is damaged." };
  let plain;
  try { plain = new TextDecoder().decode(await crypto.subtle.decrypt({ name: "AES-GCM", iv }, await codeKey(code, salt, w.rounds), data)); }
  catch (e) { return { ok: false, problem: "Wrong team code.", wrongCode: true }; }   // a wrong code (or a changed file) fails the check
  let person;
  try { person = cleanPerson(JSON.parse(plain)); } catch (e) { person = null; }
  return person ? { ok: true, person } : { ok: false, problem: "This team file is damaged." };
}

// One person's data as the Team tab keeps it: counts only, never what went wrong
function cleanPerson(p) {
  if (!p || typeof p !== "object") return null;
  const name = cleanName(p.name), email = cleanEmail(p.email);
  if (!name || !validEmail(email)) return null;
  const reports = (Array.isArray(p.reports) ? p.reports : []).slice(0, 5000).map(r => {
    if (!r || typeof r !== "object" || r.example === true) return null;
    const type = OLD_MISTAKE_TYPES[r.type] || r.type;
    return SECTIONS.includes(r.section) && MISTAKE_TYPES.includes(type) ? { date: isISO(r.date) ? r.date : "", section: r.section, type } : null;
  }).filter(Boolean);
  const steps = Array.isArray(p.steps) && p.steps.length === START_STEPS.length ? p.steps.map(v => v === true) : START_STEPS.map(() => false);
  const pilots = (Array.isArray(p.pilots) ? p.pilots : []).slice(0, 50).map(cleanPilot).filter(Boolean)
    .map(x => ({ baselineMinutes: x.baselineMinutes, aiMinutes: x.aiMinutes, errorsCaught: x.errorsCaught, startedWith: x.startedWith, stillUsing: x.stillUsing }));
  return { name, email, reports, steps, pilots, exportedOn: isISO(p.exportedOn) ? p.exportedOn : "", demo: p.demo === true };
}

}

// The roster: name,email,teams. Quotes are allowed ("Smith, Jo"); a header row is optional.
function parseCsv(text) {
  const rows = []; let row = [], cell = "", quoted = false;
  for (let i = 0; i < text.length; i++) {
    const c = text[i];
    if (quoted) { if (c !== '"') cell += c; else if (text[i + 1] === '"') { cell += '"'; i++; } else quoted = false; }
    else if (c === '"') quoted = true;
    else if (c === ",") { row.push(cell); cell = ""; }
    else if (c === "\n" || c === "\r") { if (c === "\r" && text[i + 1] === "\n") i++; row.push(cell); rows.push(row); row = []; cell = ""; }
    else cell += c;
  }
  if (cell || row.length) { row.push(cell); rows.push(row); }
  return rows.filter(r => r.some(x => x.trim()));
}
async function readRoster(file) {
  if (!file || !isManager()) return;
  let text = "";
  try { text = file.size > 1000000 ? "" : await file.text(); } catch (e) { text = ""; }
  const rows = parseCsv(text.replace(/^﻿/, "")), head = (rows[0] || []).map(h => h.trim().toLowerCase());
  const col = (word, fallback) => head.includes(word) ? head.indexOf(word) : fallback, hasHead = head.includes("email");
  const [n, e, t] = [col("name", 0), col("email", 1), col("teams", head.includes("team") ? head.indexOf("team") : 2)];
  const byEmail = {}; let skipped = 0;
  rows.slice(hasHead ? 1 : 0, 5000).forEach(r => {
    const email = cleanEmail(r[e]);
    if (!validEmail(email)) { skipped++; return; }
    byEmail[email] = { name: cleanName(r[n]), email, teams: String(r[t] || "").split(";").map(cleanTeam).filter(Boolean).slice(0, 10) };
  });
  const list = Object.keys(byEmail).map(k => byEmail[k]);
  if (!list.length) { tell("roster", "No one could be read from that file. It needs the columns name, email, teams."); draw(); return; }
  save("roster", list); teamPick = -1;
  const teams = new Set(list.flatMap(r => r.teams)).size;
  tell("roster", "Roster: " + persons(list.length) + " in " + plural(teams, "team") + "." + (skipped ? " " + plural(skipped, "row") + " skipped (no valid email)." : ""));
  draw(); focusId("rosterFile");
}

async function importTeam() {   // open each file with the code; the same email replaces that person's old entry
  if (!canTeam()) return;
  const box = document.getElementById("importMsg"), files = Array.from(document.getElementById("importFiles").files || []), code = cleanCode(importCode);
  const done = html => { importNote = html; draw(); focusId("importBtn"); };
  if (!canEncrypt()) return done("<p>This browser can't open locked files. Try an up-to-date Edge or Chrome.</p>");
  if (code.length < 8) { box.innerHTML = "<p>Type the team code first.</p>"; focusId("importCode"); return; }
  if (!files.length) { box.innerHTML = "<p>Pick the files first.</p>"; focusId("importFiles"); return; }
  box.innerHTML = "<p>Opening " + plural(files.length, "file") + "…</p>";
  const list = loadTeam(), problems = [];
  let added = 0, updated = 0;
  for (const f of files.slice(0, 500)) {
    let text = "";
    try { text = await f.text(); } catch (e) { problems.push([f.name, "This file couldn't be read."]); continue; }
    const r = await openTeamFile(text, code);
    if (!r.ok) { problems.push([f.name, r.problem]); continue; }
    const at = list.findIndex(p => p.email === r.person.email);
    if (at >= 0) { list[at] = r.person; updated++; } else { list.push(r.person); added++; }
  }
  const n = added + updated, wrong = problems.length && problems.every(p => p[1] === "Wrong team code.");
  if (n) saveTeam(list);
  done((n ? `<p><b>Imported ${persons(n)}${updated ? ` (${updated} updated)` : ""}.</b></p>` : `<p><b>${wrong ? "Wrong team code." : "Nothing imported."}</b></p>`) +
    (problems.length && !(wrong && !n) ? `<ul class="problems">${problems.map(p => `<li>${safe(p[0])}: ${p[1]}</li>`).join("")}</ul>` : "") +
    (wrong && !n ? "<p class='small'>Check the code with whoever made it. Capitals, spaces and dashes don't matter.</p>" : ""));
}

function csvCell(v) {       // one CSV cell: quoted if needed; a leading = + - @ can't start a spreadsheet formula
  let s = String(v);
  if (/^[=+\-@]/.test(s)) s = "'" + s;
  return /[",\n\r]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
}
