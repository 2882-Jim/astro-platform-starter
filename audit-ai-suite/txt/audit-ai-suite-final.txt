// ---------- Saving (only in this browser, under names starting "aiSuite.") ----------
const STORE = "aiSuite.";
const memory = {};          // a copy in memory, so the page still works if the browser blocks saving
let storageWorks = true;

try { localStorage.setItem(STORE + "test", "1"); localStorage.removeItem(STORE + "test"); }
catch (e) { storageWorks = false; }

function load(key, fallback) {      // read a saved value; never breaks the page
  if (key in memory) return memory[key];
  try {
    const text = localStorage.getItem(STORE + key);
    const value = text === null ? fallback : JSON.parse(text);
    memory[key] = value;
    return value;
  } catch (e) { storageWorks = false; return fallback; }
}

function save(key, value) {         // save a value; returns false if the browser wouldn't save it
  memory[key] = value;
  try { localStorage.setItem(STORE + key, JSON.stringify(value)); return true; }
  catch (e) { storageWorks = false; return false; }
}

function clearSaved() {             // delete everything this app saved in this browser
  Object.keys(memory).forEach(k => delete memory[k]);
  try {
    Object.keys(localStorage).filter(k => k.startsWith(STORE)).forEach(k => localStorage.removeItem(k));
    return true;
  } catch (e) { storageWorks = false; return false; }
}

function loadList(key) {            // a saved list, or an empty one if it's missing or damaged
  const list = load(key, []);
  return Array.isArray(list) ? list : [];
}

// ---------- Text and labels ----------
function safe(text) {               // stops typed or loaded text being treated as page code
  return String(text == null ? "" : text).replace(/&/g, "&amp;").replace(/</g, "&lt;")
    .replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}

function label(rating) {            // a coloured rating label with the word on it
  return RATINGS[rating] ? `<span class="label ${RATINGS[rating].css}">${rating}</span>` : safe(rating);
}

function options(list, selected) {  // turns a list into dropdown options (value = position in the list)
  return list.map((item, i) => `<option value="${i}"${i === selected ? " selected" : ""}>${safe(item)}</option>`).join("");
}

function cut(text, max) {           // trims typed text to a sensible length
  return typeof text === "string" ? text.trim().slice(0, max) : "";
}

function matches(text, typed) {     // search: ignores capitals; every word typed must appear
  const words = String(typed).toLowerCase().split(/\s+/).filter(w => w);
  return words.length > 0 && words.every(w => text.toLowerCase().includes(w));
}

function privacyHint(text) {        // a gentle warning if the text might name someone
  const t = String(text || "");
  const word = new RegExp("\\b(" + PRIVACY_WORDS.join("|") + ")\\b", "i");
  return (/\S+@\S+/.test(t) || word.test(t) || /\d{8,}/.test(t))
    ? "This might include a name, email address or account number. Remove anything that identifies a client or a person." : "";
}

// ---------- Dates (saved as YYYY-MM-DD, shown as 6 Nov 2026) ----------
const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
function pad(n) { return String(n).padStart(2, "0"); }
function todayISO() { const d = new Date(); return d.getFullYear() + "-" + pad(d.getMonth() + 1) + "-" + pad(d.getDate()); }
function dayNo(iso) { const p = iso.split("-").map(Number); return Date.UTC(p[0], p[1] - 1, p[2]) / 864e5; }  // whole days since 1970
function addDays(iso, n) { const d = new Date((dayNo(iso) + n) * 864e5); return d.getUTCFullYear() + "-" + pad(d.getUTCMonth() + 1) + "-" + pad(d.getUTCDate()); }
function isISO(s) { return typeof s === "string" && /^\d{4}-\d{2}-\d{2}$/.test(s) && addDays(s, 0) === s; }
function daysBetween(a, b) { return dayNo(b) - dayNo(a); }
function weekNo(iso) { return Math.floor((dayNo(iso) - dayNo("2026-01-05")) / 7); }   // weeks since Monday 5 Jan 2026
function formatDate(iso) { if (!isISO(iso)) return ""; const p = iso.split("-").map(Number); return p[2] + " " + MONTHS[p[1] - 1] + " " + p[0]; }
function newId(prefix) { return prefix + "-" + Date.now().toString(36) + "-" + Math.random().toString(36).slice(2, 6); }

  if (lvl < need) return { key: "notyet", say: need === 3 ? "Not yet, ask your manager" : "Not yet, ask your senior", why: "Needs " + task.minLevel + " or above." };
  if (lvl <= HAND_FIRST_UP_TO[rating]) return { key: "byhand", say: "Do it by hand first",

  return { key: "goahead", say: "Go ahead", why: "Follow the rating's 'what to do'." };
}

  if (onFile) return { rating: "Medium", reason: "It goes on file or is relied on, so every item must be checked to source." };
  return { rating: "Low", reason: clientData ? "Nothing relies on it. Client data is only fine here because the tool is firm-approved."
                                           : "It only helps you think or draft, and nothing relies on it." };
}

// ---------- Saved reports (checked every time they're read, so bad data never breaks a page) ----------
function cleanReport(r) {
  if (!r || typeof r !== "object" || !SECTIONS.includes(r.section) || !MISTAKE_TYPES.includes(r.type)) return null;
  const wrong = cut(r.wrong, 300);
  if (!wrong) return null;
  return { id: typeof r.id === "string" && r.id.length <= 40 ? r.id : (r.date + r.section + r.type + wrong).slice(0, 40),
           date: isISO(r.date) ? r.date : todayISO(), section: r.section, type: r.type, wrong,
           caught: cut(r.caught, 200), impact: cut(r.impact, 200),
           context: r.context === "practice" || r.context === "real" ? r.context : "",
           task: TASKS.some(t => t.task === r.task) ? r.task : "", example: r.example === true };
}
function loadReports() { return loadList("reports").map(cleanReport).filter(Boolean); }

function loadSteps() { const s = load("steps", null); return Array.isArray(s) && s.length === START_STEPS.length ? s.map(v => v === true) : START_STEPS.map(() => false); }
function loadPicks() { const p = load("picks", null); return validPicks(p) || PITFALLS.map(() => -1); }
function validPicks(p) {
  return Array.isArray(p) && p.length === PITFALLS.length &&
    p.every((v, i) => Number.isInteger(v) && v >= -1 && v < PITFALLS[i].sentences.length) ? p : null;
}

// ---------- The file staff send to their manager (no names in it) ----------
function getAnonId() {             // a random code so a manager can tell files from the same person apart
  let id = load("anonId", "");
  if (!/^[a-z0-9-]{4,40}$/.test(id)) { id = "p-" + Math.random().toString(36).slice(2, 10); save("anonId", id); }
  return id;
}

function makeExport(madeWith) {    // madeWith = "basic" or "final"
  const out = { app: EXPORT_APP, version: EXPORT_VERSION, madeWith, exportedOn: todayISO(), anonId: getAnonId(),
    level: load("levelSet", false) ? (LEVELS[load("level", 0)] || "") : "",
    reports: loadReports().filter(r => !r.example).map(r => ({ id: r.id, date: r.date, section: r.section, type: r.type,
      wrong: r.wrong, caught: r.caught, impact: r.impact, context: r.context, task: r.task })) };
  if (madeWith === "final") {
    out.steps = loadSteps();
    out.game = { picks: loadPicks() };
    out.settingsCopiedOn = load("settingsCopiedOn", "");
    out.notes = loadList("notes").map(cleanNote).filter(Boolean);
    out.pilots = loadList("pilots").map(cleanPilot).filter(Boolean);
  }
  return out;
}

function cleanNote(n) {            // an AI use note: reviewer role only, never a name
  if (!n || typeof n !== "object" || !isISO(n.date)) return null;
  return { id: typeof n.id === "string" ? n.id.slice(0, 40) : "", date: n.date,
           task: TASKS.some(t => t.task === n.task) ? n.task : "Other",
           tool: APPROVED_TOOLS.some(t => t.name === n.tool) ? n.tool : "Other",
           checked: BEFORE_YOU_USE.map((c, i) => Array.isArray(n.checked) && n.checked[i] === true),
           reviewerRole: LEVELS.includes(n.reviewerRole) ? n.reviewerRole : "" };
}

function cleanPilot(p) {           // a pilot: whole numbers in sensible ranges, or it's skipped
  const whole = (v, lo, hi) => Number.isInteger(v) && v >= lo && v <= hi;
  if (!p || typeof p !== "object" || !whole(p.baselineMinutes, 0, 600) || !whole(p.aiMinutes, 0, 600) ||
      !whole(p.errorsCaught, 0, 1000) || !whole(p.startedWith, 1, 500) || !whole(p.stillUsing, 0, p.startedWith)) return null;
  return { id: typeof p.id === "string" ? p.id.slice(0, 40) : "", goal: cut(p.goal, 150),
           tool: APPROVED_TOOLS.some(t => t.name === p.tool) ? p.tool : "Other",
           baselineMinutes: p.baselineMinutes, aiMinutes: p.aiMinutes, errorsCaught: p.errorsCaught,
           startedWith: p.startedWith, stillUsing: p.stillUsing };
}

function normaliseExport(obj) {    // check a loaded file; bad parts are skipped, never trusted
  if (!obj || typeof obj !== "object") return { ok: false, problem: "This isn't a file from this app." };
  if (obj.app !== EXPORT_APP) return { ok: false, problem: "This file wasn't made by this app." };
  if (!Number.isInteger(obj.version) || obj.version < 1) return { ok: false, problem: "This file's version isn't recognised." };
  if (typeof obj.anonId !== "string" || !/^[a-z0-9-]{4,40}$/.test(obj.anonId)) return { ok: false, problem: "This file is missing its reference code." };
  let skipped = 0;
  const clean = (list, max, fn) => {
    if (!Array.isArray(list)) return null;
    const out = list.slice(0, max).map(fn).filter(Boolean);
    skipped += Math.min(list.length, max) - out.length;
    return out;
  };
  const reports = clean(obj.reports, 1000, r => { const c = cleanReport(r); return c && !c.example ? c : null; }) || [];
  const steps = Array.isArray(obj.steps) && obj.steps.length === START_STEPS.length && obj.steps.every(v => typeof v === "boolean") ? obj.steps : null;
  return { ok: true, skipped, newer: obj.version > EXPORT_VERSION, person: {
    anonId: obj.anonId, madeWith: obj.madeWith === "final" ? "final" : "basic",
    exportedOn: isISO(obj.exportedOn) ? obj.exportedOn : "", level: LEVELS.includes(obj.level) ? obj.level : "",
    reports, steps, picks: validPicks(obj.game && obj.game.picks),
    settingsCopiedOn: isISO(obj.settingsCopiedOn) ? obj.settingsCopiedOn : ("settingsCopiedOn" in obj ? "" : null),
    notes: clean(obj.notes, 500, cleanNote), pilots: clean(obj.pilots, 50, cleanPilot) } };
}

function readExport(text) {        // text from a file -> checked person (or a plain-English problem)
  if (typeof text !== "string" || text.length > 1000000) return { ok: false, problem: "This file is too big to be from this app." };
  try { return normaliseExport(JSON.parse(text)); }
  catch (e) { return { ok: false, problem: "This file couldn't be read. Is it the file from 'Send to my manager'?" }; }
}

function mergePeople(people) {     // one entry per person: each part comes from their latest file that has it
  const byId = {};
  people.forEach(p => (byId[p.anonId] = byId[p.anonId] || []).push(p));
  return Object.values(byId).map(files => {
    files.sort((a, b) => (a.exportedOn || "").localeCompare(b.exportedOn || ""));
    const latest = key => { for (let i = files.length - 1; i >= 0; i--) if (files[i][key] !== null && files[i][key] !== undefined) return files[i][key]; return null; };
    const seen = {}, reports = [];
    files.forEach(f => f.reports.forEach(r => { if (!seen[r.id]) { seen[r.id] = true; reports.push(r); } }));
    return { anonId: files[0].anonId, madeWith: files.some(f => f.madeWith === "final") ? "final" : "basic",
             exportedOn: files[files.length - 1].exportedOn, level: latest("level") || "", reports,
             steps: latest("steps"), picks: latest("picks"), settingsCopiedOn: latest("settingsCopiedOn"),
             notes: latest("notes"), pilots: latest("pilots"), files: files.length };
  });
}

// ---------- Heat map (used by the staff and management files) ----------
function countReports(reports) {    // how many reports in each section x mistake type square
  const counts = SECTIONS.map(() => MISTAKE_TYPES.map(() => 0));
  reports.forEach(r => counts[SECTIONS.indexOf(r.section)][MISTAKE_TYPES.indexOf(r.type)]++);
  return counts;
}
function topSquares(counts) {       // the busiest square(s): { max, names }
  let max = 0, names = [];
  counts.forEach((row, s) => row.forEach((n, t) => {
    const name = SECTIONS[s] + " / " + MISTAKE_TYPES[t];
    if (n > max) { max = n; names = [name]; } else if (n === max && n > 0) names.push(name);
  }));
  return { max, names };
}
function heatTable(counts, pickFn) { // the grid: each square is a button; darker means more reports
  const max = topSquares(counts).max;
  let html = `<table class="grid heat"><caption>AI mistakes by audit section and type</caption>
    <tr><th scope="col">Section</th>${MISTAKE_TYPES.map(t => `<th scope="col">${t}</th>`).join("")}</tr>`;
  counts.forEach((row, s) => {
    html += `<tr><th scope="row">${SECTIONS[s]}</th>` + row.map((n, t) =>
      `<td class="cellwrap h${n ? Math.ceil(4 * n / max) : 0}"><button class="cell" id="cell-${s}-${t}" data-s="${s}" data-t="${t}"
        aria-label="${SECTIONS[s]}, ${MISTAKE_TYPES[t]}: ${n} report${n === 1 ? "" : "s"}" onclick="${pickFn}(${s}, ${t})">${n}</button></td>`).join("") + `</tr>`;
  });
  return `<div class="scroll">${html}</table></div>`;
}

async function copyText(boxId, msgId, onCopied) {   // copy a box's text; if the browser blocks it, select the text instead
  const box = document.getElementById(boxId), msg = document.getElementById(msgId);
  try { await navigator.clipboard.writeText(box.value); msg.textContent = "Copied."; }
  catch (e) { box.focus(); box.select(); box.setSelectionRange(0, box.value.length); msg.textContent = "Press Ctrl+C (or Cmd+C on a Mac) to copy."; }
  if (onCopied) onCopied();
}

function downloadFile(name, text) { // save a file to the computer; returns false if the browser blocks it
  try {
    const url = URL.createObjectURL(new Blob([text], { type: "application/json" }));
    const a = document.createElement("a");
    a.href = url; a.download = name;
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 10000);
    return true;
  } catch (e) { return false; }
}

// ---------- Data check: lists any typos in the data above, in one box on the page ----------
function checkData() {
  const problems = [];
  TASKS.forEach(t => {
    const r = t.ratings || {};
    if (!SECTIONS.includes(t.section) || !RATINGS[r.personal] || !RATINGS[r.cloud] || !RATINGS[r.local] || !LEVELS.includes(t.minLevel)
        || !PROMPT_JOBS.some(j => j.name === t.job) || !PITFALLS.some(p => p.id === t.spot)) problems.push("Task: " + t.task);
  });
  PITFALLS.forEach(p => {
    if (!MISTAKE_TYPES.includes(p.type) || !(p.wrong >= 0 && p.wrong < p.sentences.length) || !p.terms.every(term => GLOSSARY.some(g => g.term === term))) problems.push("Spot the mistake: " + p.name);
  });
  APPROVED_TOOLS.forEach(t => { if (!SWITCH_OPTIONS[t.where] || !TOOL_STATUSES[t.status]) problems.push("Tool: " + t.name); });
  if (PROMPT_PRESETS.length !== LEVELS.length) problems.push("PROMPT_PRESETS needs one line per level");
  EXAMPLE_REPORTS.forEach((e, i) => { if (!SECTIONS[e[0]] || !MISTAKE_TYPES[e[1]]) problems.push("Example report " + (i + 1)); });
  return problems.length ? `<div class="error" role="alert"><b>Check the spelling in the data:</b><ul>${problems.map(p => `<li>${safe(p)}</li>`).join("")}</ul></div>` : "";
}

function weekHtml() {       // two tasks and one pitfall for this level and switch, changing every Monday
  const w = weekNo(todayISO()), wrap = (n, size) => ((n % size) + size) % size;
  const pool = TASKS.map((t, i) => i).filter(i => ["goahead", "byhand"].includes(verdictFor(TASKS[i], level, where).key) && TASKS[i].ratings[where] !== "Significant");
  const picks = pool.length ? [...new Set([pool[wrap(2 * w, pool.length)], pool[wrap(2 * w + 1, pool.length)]])] : [];

      || "<p>No tasks to suggest at this level and setting.</p>"}
    <div class="box"><p id="watchOut"><b>Watch out for:</b> ${PITFALLS[pit].name}. ${PITFALLS[pit].check}</p>

function searchHtml() {
  return `<div class="field"><label for="search">Search all tasks</label>
    <input id="search" type="search" placeholder="e.g. payroll, invoices, bank" autocomplete="off" oninput="search(this.value)"></div>
    <div id="results" aria-live="polite"></div>`;
}
function search(typed) {    // ignores capitals; every word typed must appear in the task or its section
  let html = "";
  TASKS.forEach((t, i) => {
    if (matches(t.task + " " + t.section, typed)) html += `<button class="list-btn" id="hit-${i}" onclick="openTask(${i})">${safe(t.task)}<br>
      <span class="small">${t.section}</span> ${label(t.ratings[where])}</button>`;
  });
  if (typed.trim() && !html) html = `<p class="small">No tasks found. Try a different word.</p>`;
  document.getElementById("results").innerHTML = html;
}
function openTask(i) {      // open a task's section and highlight its card
  sectionNo = SECTIONS.indexOf(TASKS[i].section);
  go("section");
  const card = document.getElementById("task-" + i);
  card.classList.add("found");
  card.scrollIntoView({ block: "center" });
}

function sectionScreen() {
  let cards = "";
  TASKS.forEach((t, i) => { if (t.section === SECTIONS[sectionNo]) cards += taskCard(t, i); });
  return `<h1>${SECTIONS[sectionNo]}</h1>${ratingsFor()}${searchHtml()}${cards}
    <div class="sectionbar">
      <button class="btn" id="secBack" onclick="moveSection(-1)">Back</button>
      <span>Section ${sectionNo + 1} of ${SECTIONS.length}</span>
      <button class="btn" id="secNext" onclick="moveSection(1)"${sectionNo === SECTIONS.length - 1 ? " disabled" : ""}>Next</button>
    </div>`;
}
function moveSection(step) {  // Back on section 1 returns to the list of sections
  if (step < 0 && sectionNo === 0) { go("tasks"); return; }
  sectionNo = Math.min(SECTIONS.length - 1, Math.max(0, sectionNo + step));
  go("section");
}

function taskCard(t, i) {   // one task: rating, reason, "for you" verdict, and "See more" with what to do next
  const rating = t.ratings[where], v = verdictFor(t, level, where);
  const reason = rating === "Not permitted" && t.ratings.cloud !== "Not permitted" ? "Client work must stay out of personal AI accounts." : t.reason;
  const link = (kind, id, text) => `<button class="plain" id="${id}-${i}" data-link="${kind}" onclick="cardLink('${kind}', ${i})">${text}</button>`;
  return `<div class="box task-card" id="task-${i}" data-task="${i}"><h2>${safe(t.task)}</h2>${label(rating)}<p>${reason}</p>
    <p class="foryou v-${v.key}"><b>For you (${LEVELS[level]}): ${v.say}.</b> ${v.why}</p>
    <button class="plain more" id="more-${i}" aria-expanded="false" aria-controls="ex-${i}" onclick="toggleMore(${i})">See more</button>
    <div class="example" id="ex-${i}" hidden><p><b>Good AI use looks like:</b> ${t.goodUse}</p><p><b>What to do:</b> ${RATINGS[rating].todo}</p>
      <div class="whatnext"><h3>What next?</h3>${link("prompt", "prompt", "Build a prompt for this")}${link("spot", "spot", "Spot a similar mistake")}${link("report", "logit", "Log a mistake like this")}${link("note", "note", "Add to my AI use note")}</div></div></div>`;
}
function toggleMore(i) {
  const button = document.getElementById("more-" + i), box = document.getElementById("ex-" + i);
  box.hidden = !box.hidden;
  button.setAttribute("aria-expanded", String(!box.hidden));
  button.textContent = box.hidden ? "See more" : "Hide";
}
function cardLink(kind, i) {   // the four "What next?" links: each opens a screen already set up for this task
  const t = TASKS[i];
  if (kind === "prompt") { Object.assign(pb, { mode: "ai", task: i, job: Math.max(0, PROMPT_JOBS.findIndex(j => j.name === t.job)), out: "" }); delete pb.picks.format; go("prompt"); }
  if (kind === "spot") { gameNo = Math.max(0, PITFALLS.findIndex(p => p.id === t.spot)); go("game"); }
  if (kind === "report") { report.section = SECTIONS.indexOf(t.section); report.task = i; go("report"); }
  if (kind === "note") { note.task = i; note.tool = toolFor(where); go("note"); }
}

  return `<p class="foryou v-${say[0]}" id="atLevel"><b>At your level (${LEVELS[level]}):</b> ${say[1]}</p>`;
}

function alwaysRules(job) { return PROMPT_ALWAYS.concat(PROMPT_ALWAYS_FOR[job.group]); }

function makePrompt() {     // settings (if not loaded) + job + notes + rules (none twice) + document
  const job = PROMPT_JOBS[pb.job], top = pb.loaded === "no" ? settingsPrompt(level) : "";
  const picked = PROMPT_CHOICES.filter(c => c.for === "all" || c.for === job.group).map(c => c.options[choiceFor(c)][1]).filter(r => r);
  const rules = alwaysRules(job).concat(picked).filter(r => !top.includes(r));
  let prompt = (top ? top + "\n\n" : "") + "Task: " + job.instruction + "\n";
  if (pb.ask.trim()) prompt += "\nMy notes, in my own words:\n" + pb.ask.trim() + "\n";
  prompt += "\nRules:\n" + rules.map(r => "- " + r).join("\n") + "\n";
  if (job.group === "doc") prompt += "\n--- DOCUMENT START ---\n" + (pb.source.trim() || "[paste your document here]") + "\n--- DOCUMENT END ---";
  else if (pb.source.trim()) prompt += "\n--- BACKGROUND START ---\n" + pb.source.trim() + "\n--- BACKGROUND END ---";
  return prompt.trim();
}
function buildPrompt() { if (!blocked()) { pb.out = makePrompt(); draw(); focusId("promptOut"); } }

function pickSentence(i) {
  const picks = loadPicks();

  if (picks.every(x => x >= 0)) tick(3);   // answering all eight ticks "Play Spot the mistake"
  draw();
  focusId("verdict");                      // the picked button is now disabled, so read out the result instead
}

function glossaryScreen() {
  const order = GLOSSARY.map((g, n) => n).sort((a, b) => GLOSSARY[a].term.localeCompare(GLOSSARY[b].term, "en", { sensitivity: "base" }));
  return `<h1>Glossary</h1><p class="grey">The words used in this app, in plain English.</p>
    <div class="field"><label for="glossSearch">Search the glossary</label>
      <input id="glossSearch" type="search" placeholder="e.g. prompt" autocomplete="off" oninput="filterGlossary(this.value)"></div>
    <p class="small" id="glossCount" role="status"></p>
    ${order.map(n => `<div class="box term" id="term-${n}"><h2>${GLOSSARY[n].term}</h2><p>${GLOSSARY[n].means}</p></div>`).join("")}`;
}
function filterGlossary(typed) {   // hides terms that don't match (no redraw, so typing isn't interrupted)
  let shown = 0;
  GLOSSARY.forEach((g, n) => { const hit = !typed.trim() || matches(g.term + " " + g.means, typed); document.getElementById("term-" + n).hidden = !hit; shown += hit; });
  document.getElementById("glossCount").textContent = !typed.trim() ? "" : shown ? "Showing " + shown + " of " + GLOSSARY.length + " terms." : "No terms found. Try a different word.";
}
function openTerm(n) {      // from Spot the mistake: open the glossary at one term, highlighted

function wholeOrNull(v) { const n = Number(v); return v !== "" && Number.isInteger(n) && n >= 0 ? n : null; }
function liveLine() {       // the difference per task, while you type
  const b = wholeOrNull(pilot.baselineMinutes), a = wholeOrNull(pilot.aiMinutes);
  if (b === null || a === null) return "Fill in both times to see the difference.";
  return a < b ? "Minutes saved per task: " + (b - a) : a > b ? "AI was " + plural(a - b, "minute") + " slower per task" : "No difference per task.";
}
function pilotResult(p) {   // e.g. "Saved 10 minutes per task (25% faster)"
  const diff = p.baselineMinutes - p.aiMinutes, pct = p.baselineMinutes ? Math.round(100 * Math.abs(diff) / p.baselineMinutes) : null;
  if (diff > 0) return "Saved " + plural(diff, "minute") + " per task (" + pct + "% faster)";
  if (diff < 0) return "AI was " + plural(-diff, "minute") + " slower per task" + (pct === null ? "" : " (" + pct + "% slower)");
  return "No difference in time per task";
}

function savePilot() {      // checked with the same rules a manager's dashboard uses (cleanPilot)
  const num = v => v === "" ? NaN : Number(v);
  const raw = { id: newId("p"), goal: cut(pilot.goal, 150), tool: APPROVED_TOOLS[pilot.tool].name, baselineMinutes: num(pilot.baselineMinutes),
    aiMinutes: num(pilot.aiMinutes), errorsCaught: num(pilot.errorsCaught), startedWith: num(pilot.startedWith), stillUsing: num(pilot.stillUsing) };
  const ok = (v, lo, hi) => Number.isInteger(v) && v >= lo && v <= hi, clean = cleanPilot(raw);
  if (!raw.goal) problems.push("Say what you're testing (one line is fine).");
  if (!ok(raw.baselineMinutes, 0, 600)) problems.push("Minutes per task without AI: a whole number from 0 to 600.");
  if (!ok(raw.aiMinutes, 0, 600)) problems.push("Minutes per task with AI: a whole number from 0 to 600.");
  if (!ok(raw.errorsCaught, 0, 1000)) problems.push("AI mistakes caught: a whole number from 0 to 1,000.");
  if (!ok(raw.startedWith, 1, 500)) problems.push("People who started: a whole number from 1 to 500.");
  if (!ok(raw.stillUsing, 0, 500)) problems.push("People still using it: a whole number from 0 to 500.");
  else if (ok(raw.startedWith, 1, 500) && raw.stillUsing > raw.startedWith) problems.push("People still using it can't be more than the people who started.");
  if (!clean && !problems.length) problems.push("Please check the numbers.");
  if (problems.length) { draw(); return; }
  const list = loadPilots();
  list.unshift(clean);
  message = save("pilots", list) ? "Pilot saved." : "Saved for now, but this browser isn't saving, so it will be lost when you close the page.";
  pilot = { goal: "", tool: pilot.tool, baselineMinutes: "", aiMinutes: "", errorsCaught: "", startedWith: "", stillUsing: "" };
  draw();
}
